cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(PicoLink C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(PicoLink main.cpp ../swimlib.c)
# Wren's sketch is held to every warning; pico-swimlib's swimlib.c is built as it is.
set_source_files_properties(main.cpp PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror;-Wno-unused-parameter")
target_link_libraries(PicoLink pico_stdlib hardware_pio hardware_dma hardware_clocks pico_time)
pico_enable_stdio_usb(PicoLink 1)
pico_enable_stdio_uart(PicoLink 0)
pico_add_extra_outputs(PicoLink)
